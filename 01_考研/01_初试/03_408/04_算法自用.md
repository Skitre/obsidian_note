#### DFS
**邻接表**
```
bool visited[MaxVnum];
void DFS(ALGraph G, int v){
	visit(v);
	visited[v]=TRUE;
	ArcNode *p = G->adjlist[v].first;
	while(p){
		int w = p->adjvex;
		if(!visited[w]){
			DFS(G,w);
		}
		p = p->next;
	}
}

## 查找与排序
#### 快速排序
```
void Qsort(int A[],int low,int high){
	if(low<high){
		int pivot = A[low];
		int i = low,j = high;
		while(i<j){
			while(i<j&&A[j]>piovt) j--;
			while(i<j&&A[i]<piovt) i++;
			if(i<j){
				int temp = A[i];
				A[i] = A[j];
				A[j] = temp;
			}
			A[low] = A[j];
			A[j] = piovt;
			Qsort(A,low,j-1);
			Qsort(A,j,high);
		}
	}
}
```
#### 二分查找
```
void binary_find(int arr[],int length,int x){
	int i = 0,j = length - 1;
	while(i<j){
		int mid = (i + j)/2;
		if(arr[mid] == x){
			return mid;
		}
		else if(arr[mid] < x)
			j = mid + 1;
		}
		else if(arr[mid] > x){
			i = mid - 1;
		}
	}
}
